37def spending_by_workday(transactions: pd.DataFrame, date: Optional[str] = None) -> DataFrame: 

38 """ 

39 Функция принимает транзакции в виде DataFrame и опциональную дату в виде строки, а возвращает данные по средним 

40 тратам в виде DataFrame. 

41 """ 

42 

43 reports_logger.info("Запуск <spending_by_workday>") 

44 if date is None: 

45 date = "31.12.2021" 

46 date_obj = datetime.datetime.strptime(date, "%d.%m.%Y") 

47 date_full_obj = date_obj.replace(hour=23, minute=59, second=59) 

48 date_border = date_obj - datetime.timedelta(days=89) 

49 

50 spent_all_df = transactions.loc[transactions.Итого < 0] 

51 spent_indexed_df = spent_all_df.reset_index(drop=True) 

52 spent_indexed_df["Итого"] = spent_indexed_df["Итого"].astype(str).transform(quantize_decimal) 

53 

54 reports_logger.info("Создание датафрейма операций за 3 месяца") 

55 sector_date_df = spent_indexed_df[ 

56 (spent_indexed_df["Дата_операции"] <= pd.Timestamp(date_full_obj)) 

57 & (pd.Timestamp(date_border) <= spent_indexed_df["Дата_операции"]) 

58 ] 

59 

60 only_day_df = sector_date_df[["Дата_операции", "Итого"]] 

61 only_day_df.rename(columns={"Дата_операции": "Дата_операций"}, inplace=True) 

62 only_day_df["Дата_операций"] = only_day_df["Дата_операций"].apply(lambda x: x.date()) 

63 if not pd.api.types.is_datetime64_any_dtype(only_day_df["Дата_операций"]): 

64 only_day_df["Дата_операций"] = pd.to_datetime(only_day_df["Дата_операций"]) 

65 only_day_df["День_недели"] = only_day_df["Дата_операций"].dt.day_name() 

66 only_day_df["Итого"] = only_day_df["Итого"].apply(lambda x: abs(x)) 

67 

68 reports_logger.info("Формирование статистики по тратам в выходной/рабочий день") 

69 grouped_df = only_day_df.groupby(["Дата_операций", "День_недели"]).agg({"Итого": "mean"}) 

70 grouped_df.reset_index(level=[0, 1], drop=False, inplace=True) 

71 grouped_df["Итого"] = grouped_df["Итого"].astype(str).transform(quantize_decimal) 

72 grouped_df["Дата_операций"] = grouped_df["Дата_операций"].astype(str) 

73 

74 reports_logger.info("Завершение <spending_by_workday>") 

75 return grouped_df
